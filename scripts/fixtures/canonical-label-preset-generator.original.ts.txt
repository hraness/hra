const verifySources = () => {
  for (const [name, source] of Object.entries(sources)) {
    const root = join(import.meta.dir, name);
    for (const [path, expected] of Object.entries(source.pins)) {
      if (sha256(readBounded(join(root, path))) !== expected) throw new Error("CAPTURE_SOURCE_MISMATCH");
    }
    const files: { path: string; sha256: string }[] = [];
    const visit = (directory: string) => {
      for (const entry of readdirSync(directory, { withFileTypes: true })) {
        if (!/^[A-Za-z0-9_.-]+$/u.test(entry.name)) throw new Error("CAPTURE_SOURCE_ENTRY_INVALID");
        const path = join(directory, entry.name);
        if (entry.isDirectory()) visit(path);
        else if (entry.isFile()) files.push({ path: relative(root, path), sha256: sha256(readBounded(path)) });
        else throw new Error("CAPTURE_SOURCE_ENTRY_INVALID");
      }
    };
    visit(join(root, "src"));
    files.sort((a, b) => a.path < b.path ? -1 : a.path > b.path ? 1 : 0);
    if (files.length !== source.sourceFileCount || sha256(JSON.stringify(files)) !== source.manifest) throw new Error("CAPTURE_SOURCE_MANIFEST_MISMATCH");
    const declared = z.object({ dependencies: z.record(z.string(), z.string()) }).passthrough()
      .parse(JSON.parse(readBounded(join(root, "package.json")).toString("utf8")));
    if (JSON.stringify(declared.dependencies) !== JSON.stringify(dependencies)) throw new Error("CAPTURE_DEPENDENCIES_MISMATCH");

const snapshot = (db: Database) => {
  const schema = z.object({ type: z.string(), name: z.string(), tbl_name: z.string(), sql: z.string().nullable() }).strict().array().max(2048)
    .parse(db.query("SELECT type,name,tbl_name,sql FROM sqlite_master ORDER BY type,name LIMIT 2049").all());
  const tables = schema.filter((row) => row.type === "table");
  if (tables.length > 256) throw new Error("CAPTURE_TABLE_LIMIT");
  const rows = Object.fromEntries(tables.map(({ name }) => {
    const entries = db.query(`SELECT * FROM ${quote(name)} LIMIT 4097`).all();
    if (entries.length > 4096) throw new Error("CAPTURE_ROW_LIMIT");
    entries.sort((a, b) => JSON.stringify(a) < JSON.stringify(b) ? -1 : JSON.stringify(a) > JSON.stringify(b) ? 1 : 0);
    return [name, entries];
  }));
  const value = { version: db.query("PRAGMA user_version").get(),
    ledger: db.query("SELECT * FROM migrations ORDER BY version").all(),
    foreignKeys: db.query("PRAGMA foreign_key_check").all(), schema, rows };
  if (value.foreignKeys.length !== 0 || Buffer.byteLength(JSON.stringify(value)) > 8 * 1024 * 1024) throw new Error("CAPTURE_SNAPSHOT_INVALID");
  return value;
};
