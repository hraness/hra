const assertPrivatePathFree = (value: string | Uint8Array) => {
  const bytes = typeof value === "string" ? Buffer.from(value, "utf8") : Buffer.from(value);
  for (const prefix of hostRoots) {
    for (const suffix of ["/", "\\", "\u0000", '"', "'"]) {
      const text = prefix + suffix;
      const littleEndian = Buffer.from(text, "utf16le");
      const bigEndian = Buffer.from(littleEndian);
      bigEndian.swap16();
      assert.equal(bytes.includes(Buffer.from(text, "utf8")), false, "Private UTF-8 path refused.");
      assert.equal(bytes.includes(littleEndian), false, "Private UTF-16LE path refused.");
      assert.equal(bytes.includes(bigEndian), false, "Private UTF-16BE path refused.");
    }
    if (typeof value === "string") assert.equal(value === prefix || value.includes(prefix + "/") || value.includes(prefix + "\\"), false);
  }
};
const assertDecodedPrivatePathFree = (value: unknown, depth = 0): void => {
  assert.ok(depth <= 64);
  if (typeof value === "string") { assertPrivatePathFree(value); return; }
  if (value === null || typeof value === "number" || typeof value === "boolean") return;
  if (Array.isArray(value)) {
    assert.ok(value.length <= 4096);
    for (const item of value) assertDecodedPrivatePathFree(item, depth + 1);
    return;
  }
  const object = record(value);
  for (const [key, item] of Object.entries(object)) {
    assertPrivatePathFree(key);
    assertDecodedPrivatePathFree(item, depth + 1);
  }
};

const readBounded = (path: string) => {
  const stat = lstatSync(path);
  assert.equal(stat.isFile(), true);
  assert.ok(stat.size <= 8 * 1024 * 1024);
  return readFileSync(path);
};
const originalDependencyManifests = new Map<string, string>();
const verifySource = (definition: typeof definitions[number]) => {
  const sourceRoot = join(root, definition.name);
  assert.equal(lstatSync(sourceRoot).isDirectory(), true);
  for (const [path, expected] of Object.entries(definition.sourceHashes)) {
    assert.equal(hash(readBounded(join(sourceRoot, path))), expected, path);
  }
  const files: { path: string; sha256: string }[] = [];
  const visit = (directory: string) => {
    for (const entry of readdirSync(directory, { withFileTypes: true })) {
      assert.match(entry.name, /^[A-Za-z0-9_.-]+$/);
      const path = join(directory, entry.name);
      if (entry.isDirectory()) visit(path);
      else if (entry.isFile()) files.push({ path: relative(sourceRoot, path), sha256: hash(readBounded(path)) });
      else throw new Error("Unexpected source entry.");
    }
  };
  visit(join(sourceRoot, "src"));
  files.sort((a, b) => a.path < b.path ? -1 : a.path > b.path ? 1 : 0);
  assert.equal(files.length, definition.sourceFileCount);
  assert.equal(hash(JSON.stringify(files)), definition.sourceManifestSha256);
  const manifest = record(JSON.parse(readBounded(join(sourceRoot, "package.json")).toString("utf8")) as unknown);
  const dependencies = record(manifest.dependencies);
  assert.deepEqual(dependencies, definition.dependencies);
  for (const [name, version] of Object.entries(definition.dependencies)) {
    const installedBytes = readBounded(join(sourceRoot, "node_modules", name, "package.json"));
    const installed = record(JSON.parse(installedBytes.toString("utf8")) as unknown);
    assert.equal(string(installed.version), version);
    const identity = `${definition.name}:${name}`;
    const digest = hash(installedBytes);
    const original = originalDependencyManifests.get(identity);
    if (original === undefined) originalDependencyManifests.set(identity, digest);
    else assert.equal(digest, original, "Archived dependency manifest changed.");
  }
  return files;
};
const snapshot = (path: string) => {
  const database = new Database(path, { create: false, strict: true });
  database.exec("PRAGMA query_only=ON");
  try {
    return database.transaction(() => {
      const schema = database.query("SELECT type,name,tbl_name,sql FROM sqlite_schema WHERE name NOT LIKE 'sqlite_%' ORDER BY type,name LIMIT 2049").all()
        .map((value: unknown) => {
          const row = record(value);
          exactKeys(row, ["type", "name", "tbl_name", "sql"]);
          assert.ok(row.sql === null || typeof row.sql === "string");
          return { type: string(row.type), name: string(row.name), tbl_name: string(row.tbl_name), sql: row.sql };
        });
      assert.ok(schema.length <= 2048);
      const tables = database.query("SELECT name FROM sqlite_schema WHERE type='table' ORDER BY name LIMIT 257").all()
        .map((value: unknown) => { const row = record(value); exactKeys(row, ["name"]); return { name: string(row.name) }; });
      assert.ok(tables.length <= 256);
      const rows: Record<string, unknown[]> = {};
      for (const { name } of tables) {
        assert.match(name, /^[A-Za-z0-9_]+$/);
        rows[name] = database.query(`SELECT * FROM "${name}" LIMIT 4097`).all().map((value: unknown) => record(value))
          .sort((a, b) => JSON.stringify(a) < JSON.stringify(b) ? -1 : JSON.stringify(a) > JSON.stringify(b) ? 1 : 0);
        assert.ok(rows[name].length <= 4096);
      }
      const version = record(database.query("PRAGMA user_version").get());
      exactKeys(version, ["user_version"]);
      assert.equal(Number.isSafeInteger(version.user_version), true);
      const result = { version, schema, rows,
        foreignKeys: database.query("PRAGMA foreign_key_check").all(),
        changes: database.query("SELECT total_changes() AS count").get() };
      assert.deepEqual(result.foreignKeys, []);
      assert.deepEqual(result.changes, { count: 0 });
      assert.ok(Buffer.byteLength(JSON.stringify(result)) < 8 * 1024 * 1024);
      assertDecodedPrivatePathFree(result);
      return result;
    }).deferred();
  } finally { database.close(false); }
};
const checkpoint = (path: string): void => {
  const database = new Database(path, { create: false, strict: true });
  try {
    const row = record(database.query("PRAGMA wal_checkpoint(TRUNCATE)").get());
    exactKeys(row, ["busy", "log", "checkpointed"]);
    assert.deepEqual(row, { busy: 0, log: 0, checkpointed: 0 });
  } finally { database.close(false); }
  if (existsSync(`${path}-wal`)) assert.equal(lstatSync(`${path}-wal`).size, 0);
};
