const readBounded = (path: string, maximumBytes = 8 * 1024 * 1024) => {
  const stat = lstatSync(path);
  assert.equal(stat.isFile(), true);
  assert.ok(stat.size <= maximumBytes);
  return readFileSync(path);
};
const verifySource = () => {
  assert.equal(lstatSync(sourceRoot).isDirectory(), true);
  for (const [path, expected] of Object.entries(definition.sourceHashes)) {
    assert.equal(sha256(readBounded(join(sourceRoot, path))), expected, path);
  }
  assert.equal(readBounded(join(sourceRoot, ".bun-version")).toString("utf8").trim(), "1.3.14");
  const files: { path: string; sha256: string }[] = [];
  const visit = (directory: string) => {
    for (const entry of readdirSync(directory, { withFileTypes: true })) {
      assert.match(entry.name, /^[A-Za-z0-9_.-]+$/);
      const path = join(directory, entry.name);
      if (entry.isDirectory()) visit(path);
      else if (entry.isFile()) files.push({ path: relative(sourceRoot, path), sha256: sha256(readBounded(path)) });
      else throw new Error("Unexpected archived source entry.");
    }
  };
  visit(join(sourceRoot, "src"));
  files.sort((a, b) => a.path < b.path ? -1 : a.path > b.path ? 1 : 0);
  assert.equal(files.length, definition.sourceFileCount);
  assert.equal(sha256(JSON.stringify(files)), definition.sourceManifestSha256);
  const manifest = JSON.parse(readBounded(join(sourceRoot, "package.json")).toString("utf8"));
  assert.deepEqual(manifest.dependencies, definition.dependencies);
  for (const [name, version] of Object.entries(definition.dependencies)) {
    const installed = JSON.parse(readBounded(join(sourceRoot, "node_modules", name, "package.json")).toString("utf8"));
    assert.equal(installed.version, version);
  }
  return files;
};
const snapshot = (path: string) => {
  const database = new Database(path, { create: false, strict: true });
  database.exec("PRAGMA query_only=ON");
  try {
    return database.transaction(() => {
      const schema = database.query("SELECT type,name,tbl_name,sql FROM sqlite_schema WHERE name NOT LIKE 'sqlite_%' ORDER BY type,name LIMIT 2049").all();
      assert.ok(schema.length <= 2048);
      const tables = database.query("SELECT name FROM sqlite_schema WHERE type='table' ORDER BY name LIMIT 257").all() as { name: string }[];
      assert.ok(tables.length <= 256);
      const rows: Record<string, unknown[]> = {};
      for (const { name } of tables) {
        assert.match(name, /^[A-Za-z0-9_]+$/);
        rows[name] = database.query(`SELECT * FROM "${name}" LIMIT 4097`).all().sort((a, b) => {
          const left = JSON.stringify(a), right = JSON.stringify(b);
          return left < right ? -1 : left > right ? 1 : 0;
        });
        assert.ok(rows[name].length <= 4096);
      }
      const result = { userVersion: database.query("PRAGMA user_version").get(), schema, rows,
        foreignKeyCheck: database.query("PRAGMA foreign_key_check").all(),
        inspectionChanges: database.query("SELECT total_changes() AS count").get() };
      assert.deepEqual(result.foreignKeyCheck, []);
      assert.deepEqual(result.inspectionChanges, { count: 0 });
      assert.ok(Buffer.byteLength(JSON.stringify(result)) <= 8 * 1024 * 1024);
      return result;
    }).deferred();
  } finally { database.close(false); }
};
