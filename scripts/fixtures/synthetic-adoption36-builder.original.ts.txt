// Parse literal templates, never eval/transpile/import archived source code.
function template(source: string, name: string, substitutions: Readonly<Record<string, string>> = {}): string {
  const file = ts.createSourceFile("frozen.ts", source, ts.ScriptTarget.Latest, true, ts.ScriptKind.TS);
  const matches: unknown[] = [];
  const visit = (node: unknown): void => {
    if (ts.isVariableDeclaration(node) && ts.isIdentifier(node.name) && node.name.text === name) matches.push(node.initializer);
    ts.forEachChild(node, visit);
  };
  visit(file);
  assert.equal(matches.length, 1, name);
  const node = matches[0];
  if (ts.isNoSubstitutionTemplateLiteral(node)) return node.text;
  assert.equal(ts.isTemplateExpression(node), true, name);
  let value = node.head.text;
  for (const span of node.templateSpans) {
    assert.equal(ts.isIdentifier(span.expression), true);
    const substitution = substitutions[span.expression.text];
    assert.notEqual(substitution, undefined, span.expression.text);
    value += substitution + span.literal.text;
  }
  return value;
}
